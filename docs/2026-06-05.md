### 1. 변수와 자료형

#### 변수 선언
| 키워드 | 특징 |
|--------|------|
| `var` | ES5. scope 범위가 불명확해 메모리 누수 발생 가능 |
| `let` | ES6. 블록 범위, 값 변경 가능 |
| `const` | ES6. 블록 범위, 값 변경 불가 (상수) |

#### 자료형
```javascript
// 기본형
let a = 10          // number
let b = 10.5        // number (정수/실수 구분 없음)
let c = 'Hello'     // string
let d = true        // boolean
let e = null        // null
let f               // undefined

// 참조형
let arr = ["aaa", "bbb"]            // array
let obj = { name: "홍길동", age: 20 } // object (JSON 형태)
```

> 자료형 확인: `typeof 변수`

#### 형변환
```javascript
Number("10")    // → 10  (문자 → 숫자)
parseInt("10")  // → 10  (문자 → 정수)
String(10)      // → "10" (숫자 → 문자)
Boolean(0)      // → false
Boolean(1)      // → true  (0, null 외엔 true)
```

### 2. 연산자

#### 산술 연산자
```javascript
// + : 덧셈 또는 문자열 결합
"5" + 3   // → "53" (문자열 결합)
5 + 3     // → 8

// / : 자바와 달리 정수끼리 나눠도 실수 반환
5 / 2     // → 2.5  (Java는 2)

// % : 결과 부호는 왼쪽 피연산자를 따름
-7 % 3    // → -1
7 % -3    // → 1
```

#### 비교 연산자
```javascript
// == : 자료형 무시하고 값만 비교 (비권장)
'10' == 10    // → true

// === : 값 + 자료형 모두 비교 (권장)
'10' === 10   // → false
10  === 10    // → true

// != vs !==  도 동일한 원리
```

#### 논리 연산자
```javascript
&&   // AND : 둘 다 true일 때 true
||   // OR  : 하나라도 true면 true
!    // NOT : true ↔ false 반전
```

#### 삼항 연산자
```javascript
// 조건 ? 값1(true) : 값2(false)
let result = age >= 18 ? "성인" : "미성년자";
// if~else와 동일한 동작
```

### 3. 제어문 - 조건문

#### if 단일 조건
```javascript
if (조건) {
    // 조건이 true일 때만 실행
}
```

#### if~else 선택 조건
```javascript
if (조건) {
    // true
} else {
    // false
}
```

#### if~else if 다중 조건
```javascript
if (조건1) {
    // 조건1 true → 실행 후 종료
} else if (조건2) {
    // 조건2 true → 실행 후 종료
} else {
    // 모든 조건 false
}
```

#### switch 선택문
```javascript
switch (값) {
    case 1:
        // 처리
        break;      // break 없으면 다음 case까지 실행됨
    case 2:
        // 처리
        break;
    default:
        // 해당 case 없을 때
}
```

### 4. 제어문 - 반복문

#### for (기본)
```javascript
//      ①초기값  ②조건  ④증가
for (let i = 1; i <= 10; i++) {
    // ③반복 수행
}
// 실행 순서: ① → ②(true) → ③ → ④ → ②(true) → ... → ②(false) 종료
// const는 사용 불가, let/var만 사용
```

#### for in — 인덱스 순회
```javascript
let names = ["홍길동", "심청이", "이순신"];

for (let i in names) {
    console.log(i);         // 0, 1, 2 (인덱스)
    console.log(names[i]);  // 값 접근
}
```

#### for of — 값 직접 순회 ***
```javascript
for (let name of names) {
    console.log(name);  // "홍길동", "심청이", "이순신"
}
```

#### forEach ***
```javascript
names.forEach(function(name) {
    console.log(name);  // 배열 값을 1개씩 자동 호출
});
```

#### map ***
```javascript
// forEach와 유사하나 새 배열을 반환
let result = names.map(function(name) {
    return name + "님";
});
// ["홍길동님", "심청이님", "이순신님"]
```

#### while
```javascript
let i = 1;
while (i <= 10) {
    // 반복 수행
    i++;
}
// 반복 횟수가 정해지지 않은 경우 사용
```

#### 반복 제어문
```javascript
break;      // 반복문 즉시 종료
continue;   // 현재 회차 건너뛰고 다음 반복으로
```
